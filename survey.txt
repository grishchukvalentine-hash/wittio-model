function img_(form, file, title) {
  const blob = UrlFetchApp.fetch(IMG + file).getBlob();
  form.addImageItem().setImage(blob).setTitle(title).setAlignment(FormApp.Alignment.CENTER).setWidth(300);
}

function choice_(form, title, values, required) {
  return form.addMultipleChoiceItem().setTitle(title).setChoiceValues(values).setRequired(required !== false);
}

function createWittioSurvey() {
  const t = T[LANG];
  const form = FormApp.create(t.title);
  form.setDescription(t.desc)
      .setCollectEmail(false)
      .setLimitOneResponsePerUser(false)
      .setAllowResponseEdits(false)
      .setShowLinkToRespondAgain(false)
      .setProgressBar(true)
      .setConfirmationMessage(t.thanks);
  try { form.setRequireLogin(false); } catch (e) {}

  // Section 1: current product
  form.addSectionHeaderItem().setTitle(t.s1).setHelpText(t.s1help);
  img_(form, 'profile.jpg', '');
  choice_(form, t.qA, t.yes4);

  // Section 2: new screens
  form.addPageBreakItem().setTitle(t.s2).setHelpText(t.s2help);
  img_(form, 'mk_s1.png', '1. ' + t.img1);
  img_(form, 'mk_s2.png', '2. ' + t.img2);
  img_(form, 'mk_s3.png', '3. ' + t.img3);

  // Section 3: questions
  form.addPageBreakItem().setTitle(t.s3);
  choice_(form, t.q1, t.yes4);
  const q2 = form.addMultipleChoiceItem().setTitle(t.q2).setRequired(true);
  q2.setChoices(t.q2c.map(v => q2.createChoice(v))).showOtherOption(true).setHelpText(t.q2other);
  choice_(form, t.q3, t.q3c);
  choice_(form, t.q4, t.yn);
  form.addParagraphTextItem().setTitle(t.q5).setRequired(false);

  // Section 4: about you
  form.addPageBreakItem().setTitle(t.s4);
  choice_(form, t.age, t.ages);
  choice_(form, t.exp, t.yn);
  choice_(form, t.jp, t.yn);

  // Results sheet
  const ss = SpreadsheetApp.create(t.sheet);
  form.setDestination(FormApp.DestinationType.SPREADSHEET, ss.getId());

  form.setAcceptingResponses(true);
  try { form.setPublished(true); } catch (e) {}

  Logger.log('LINK FOR RESPONDENTS: ' + form.getPublishedUrl());
  try { Logger.log('SHORT LINK: ' + form.shortenFormUrl(form.getPublishedUrl())); } catch (e) {}
  Logger.log('EDIT THE FORM: ' + form.getEditUrl());
  Logger.log('RESULTS SHEET: ' + ss.getUrl());
}
